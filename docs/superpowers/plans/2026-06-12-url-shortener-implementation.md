# URL Shortener Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the approved account-based URL shortener with Next.js, Auth.js, Prisma, PostgreSQL, custom slugs, expiration, QR codes, table-first dashboard, and click analytics.

**Architecture:** Implement a single Next.js App Router monolith. Keep business logic in focused service modules under `src/server/*`, UI in `src/components/*`, route handlers in `src/app/*`, and Prisma as the database boundary. Redirect analytics records best-effort events without blocking redirect success.

**Tech Stack:** Next.js App Router, TypeScript, Tailwind CSS, Prisma, PostgreSQL, Auth.js/NextAuth, Vitest, Testing Library, Playwright, Zod, bcryptjs, nanoid, qrcode, ua-parser-js.

```ts
export function normalizeDestinationUrl(input: string): string {
  const trimmed = input.trim();

  if (!/^https?:\/\//i.test(trimmed)) {
    throw new Error("URL must include http:// or https://");
  }

  let url: URL;
  try {
    url = new URL(trimmed);
  } catch {
    throw new Error("URL is invalid");
  }

  if (url.protocol !== "http:" && url.protocol !== "https:") {
    throw new Error("Only http and https URLs are allowed");
  }

  url.hash = "";
  return url.toString();
}
```

```ts
import crypto from "node:crypto";
import { UAParser } from "ua-parser-js";

export type ClickMetadata = {
  referrer: string | null;
  country: string | null;
  device: string | null;
  browser: string | null;
  ipHash: string | null;
  userAgent: string | null;
};

export function hashIp(ip: string, secret: string): string {
  return crypto.createHmac("sha256", secret).update(ip).digest("hex");
}

export function extractClickMetadata(headers: Headers, ipHashSecret: string): ClickMetadata {
  const userAgent = headers.get("user-agent");
  const parsed = new UAParser(userAgent ?? undefined).getResult();
  const forwardedFor = headers.get("x-forwarded-for");
  const ip = forwardedFor?.split(",")[0]?.trim() || headers.get("x-real-ip");

  return {
    referrer: headers.get("referer"),
    country: headers.get("x-vercel-ip-country"),
    device: parsed.device.type ?? "desktop",
    browser: parsed.browser.name ?? null,
    ipHash: ip ? hashIp(ip, ipHashSecret) : null,
    userAgent,
  };
}
```

```ts
const blockedHostnames = new Set(["localhost", "127.0.0.1", "0.0.0.0"]);
const privateIpPatterns = [/^10\./, /^192\.168\./, /^172\.(1[6-9]|2\d|3[0-1])\./];

export function assertUrlAllowed(destinationUrl: string): void {
  const url = new URL(destinationUrl);
  const hostname = url.hostname.toLowerCase();

  if (blockedHostnames.has(hostname) || privateIpPatterns.some((pattern) => pattern.test(hostname))) {
    throw new Error("Destination URL is blocked");
  }
}
```

```ts
import { prisma } from "@/lib/prisma";

export async function consumeRateLimit(params: {
  key: string;
  scope: string;
  limit: number;
  windowMs: number;
  now?: Date;
}): Promise<void> {
  const now = params.now ?? new Date();
  const expiresAt = new Date(now.getTime() + params.windowMs);

  const current = await prisma.rateLimit.findUnique({
    where: { key_scope: { key: params.key, scope: params.scope } },
  });

  if (!current || current.expiresAt <= now) {
    await prisma.rateLimit.upsert({
      where: { key_scope: { key: params.key, scope: params.scope } },
      create: { key: params.key, scope: params.scope, count: 1, windowStart: now, expiresAt },
      update: { count: 1, windowStart: now, expiresAt },
    });
    return;
  }

  if (current.count >= params.limit) {
    throw new Error("Rate limit exceeded");
  }

  await prisma.rateLimit.update({
    where: { key_scope: { key: params.key, scope: params.scope } },
    data: { count: { increment: 1 } },
  });
}
```

```ts
      if (session.user && typeof token.id === "string") {
        session.user.id = token.id;
      }
      return session;
    },
  },
  providers: [
    Credentials({
      credentials: {
        email: {},
        password: {},
      },
      async authorize(rawCredentials) {
        const credentials = credentialsSchema.parse(rawCredentials);
        const user = await prisma.user.findUnique({ where: { email: credentials.email.toLowerCase() } });

        if (!user) {
          return null;
        }

        const isValid = await bcrypt.compare(credentials.password, user.passwordHash);
        if (!isValid) {
          return null;
        }

        return { id: user.id, email: user.email };
      },
    }),
  ],
} satisfies NextAuthConfig;

export const { handlers, auth, signIn, signOut } = NextAuth(authConfig);

export async function requireUserId(): Promise<string> {
  const session = await auth();
  const userId = session?.user?.id;

  if (!userId) {
    throw new Error("Unauthorized");
  }

  return userId;
}
```

```tsx
  });
```

```ts
  const passwordHash = await bcrypt.hash(parsed.data.password, 12);

        passwordHash,

    });
```

```ts
import { prisma } from "@/lib/prisma";

export type RedirectResolution =
  | { status: "not_found" }
  | { status: "inactive" }
  | { status: "expired" }
  | { status: "ok"; destinationUrl: string; shortLinkId: string };

type RedirectRecord = {
  id: string;
  originalUrl: string;
  isActive: boolean;
  expiresAt: Date | null;
};

export function isExpired(expiresAt: Date | null, now = new Date()): boolean {
  return expiresAt !== null && expiresAt.getTime() <= now.getTime();
}

export function resolveRedirectFromRecord(link: RedirectRecord, now = new Date()): RedirectResolution {
  if (!link.isActive) {
    return { status: "inactive" };
  }

  if (isExpired(link.expiresAt, now)) {
    return { status: "expired" };
  }

  return { status: "ok", destinationUrl: link.originalUrl, shortLinkId: link.id };
}

export async function resolveRedirect(slug: string, now = new Date()): Promise<RedirectResolution> {
  const link = await prisma.shortLink.findUnique({
    where: { slug },
    select: { id: true, originalUrl: true, isActive: true, expiresAt: true },
  });

  if (!link) {
    return { status: "not_found" };
  }

  return resolveRedirectFromRecord(link, now);
}
```

```tsx
"use client";

import { useState } from "react";
import { Button } from "@/components/ui/button";

export function CreateLinkSheet() {
  const [error, setError] = useState<string | null>(null);

  async function submit(formData: FormData) {
    setError(null);
    const response = await fetch("/api/links", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        originalUrl: formData.get("originalUrl"),
        customSlug: formData.get("customSlug") || null,
        title: formData.get("title") || null,
        expiresAt: formData.get("expiresAt") || null,
      }),
    });

    if (!response.ok) {
      setError("Unable to create link. Check the URL and slug.");
      return;
    }

    window.location.reload();
  }

  return (
    <form action={submit} className="grid gap-3 rounded-md border bg-white p-4">
      <Input name="originalUrl" placeholder="https://example.com/long-url" required />
      <Input name="customSlug" placeholder="custom-slug" />
      <Input name="title" placeholder="Title" />
      <Input name="expiresAt" type="datetime-local" />
```

```tsx
import { redirect } from "next/navigation";
import { auth } from "@/lib/auth";
import { shortUrlForSlug } from "@/lib/paths";
import { getDashboardMetrics } from "@/server/analytics/service";
import { listLinks } from "@/server/links/service";
import { DashboardShell } from "@/components/dashboard/dashboard-shell";

export default async function DashboardPage() {
  const session = await auth();

  if (!session?.user?.id) {
    redirect("/login");
  }

  const [metrics, records] = await Promise.all([
    getDashboardMetrics(session.user.id),
    listLinks(session.user.id),
  ]);

  const links = records.map((link) => ({
    id: link.id,
    originalUrl: link.originalUrl,
    shortUrl: shortUrlForSlug(link.slug),
    slug: link.slug,
    title: link.title,
    isActive: link.isActive,
    expiresAt: link.expiresAt?.toISOString() ?? null,
    createdAt: link.createdAt.toISOString(),
    clickCount: link._count.clicks,
  }));

  return <DashboardShell metrics={metrics} links={links} />;
}
```
